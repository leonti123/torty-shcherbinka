/**
 * «Торты в Щербинке»: режим мастера: вход мастера по паролю и сохранение правок сайта в GitHub.
 * Сайт шлёт POST (text/plain, JSON): {a:"login", pw} или {a:"save", token, base, force, data, files}.
 * Свойство скрипта GH_TOKEN: fine-grained токен GitHub только на репозиторий REPO, право Contents: Read and write.
 * SECRET для подписи входа создаётся сам при первом входе.
 */
const REPO = "leonti123/torty-shcherbinka", DIR = "", BRANCH = "main";
const P = p => DIR ? DIR + "/" + p : p;
const PW_SALT = "__SALT__";
const PW_HASH = "__HASH__";
const TOKEN_DAYS = 30, MAX_FAILS = 10, MAX_JS = 900000, MAX_FILES = 40, MAX_B64 = 3000000;

function doGet() { return out({ok: true, v: 1}); }

function doPost(e) {
  let req;
  try { req = JSON.parse(e.postData.contents); } catch (x) { return out({ok: false, err: "bad_json"}); }
  if (!req || typeof req !== "object") return out({ok: false, err: "bad_json"});
  try {
    if (req.a === "login") return out(login(req));
    if (req.a === "check") return out({ok: checkToken(req.token)});
    if (req.a === "save") return out(save(req));
    return out({ok: false, err: "bad_action"});
  } catch (x) {
    console.error(x);
    return out({ok: false, err: "server", msg: String(x && x.message || x).slice(0, 300)});
  }
}

function out(o) { return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON); }
function hex(bytes) { return bytes.map(function (b) { return ((b + 256) % 256).toString(16).padStart(2, "0"); }).join(""); }
function sha(s) { return hex(Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, s, Utilities.Charset.UTF_8)); }
function secret() {
  const p = PropertiesService.getScriptProperties();
  let s = p.getProperty("SECRET");
  if (!s) { s = Utilities.getUuid() + Utilities.getUuid(); p.setProperty("SECRET", s); }
  return s;
}
function sign(s) { return Utilities.base64EncodeWebSafe(Utilities.computeHmacSha256Signature(s, secret())).replace(/=+$/, ""); }

function login(req) {
  const lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    const c = CacheService.getScriptCache(), fails = +(c.get("fails") || 0);
    if (fails >= MAX_FAILS) return {ok: false, err: "locked"};
    const pw = req.pw;
    if (typeof pw === "string" && pw.length <= 100 && sha(PW_SALT + pw) === PW_HASH) {
      const exp = Date.now() + TOKEN_DAYS * 864e5, body = "m." + exp;
      c.remove("fails");
      return {ok: true, token: body + "." + sign(body), exp: exp};
    }
    c.put("fails", String(fails + 1), 900);
  } finally {
    lock.releaseLock();
  }
  Utilities.sleep(700);
  return {ok: false, err: "bad_password"};
}
function checkToken(t) {
  const parts = typeof t === "string" && t.match(/^(m\.(\d{13}))\.([A-Za-z0-9_-]{43})$/);
  return !!parts && +parts[2] > Date.now() &&
    !!PropertiesService.getScriptProperties().getProperty("SECRET") && sign(parts[1]) === parts[3];
}

function save(req) {
  if (!checkToken(req.token)) return {ok: false, err: "auth"};
  const tok = PropertiesService.getScriptProperties().getProperty("GH_TOKEN");
  if (!tok) return {ok: false, err: "no_gh_token"};
  const data = req.data;
  const lists = ["fills", "items", "wedding", "prices", "zones", "reviews", "heroPhotos", "aboutPhotos"];
  if (!data || data.v !== 1 || !data.t || Array.isArray(data.t) || typeof data.t !== "object" ||
      !data.contacts || Array.isArray(data.contacts) || typeof data.contacts !== "object" ||
      !lists.every(function (key) { return Array.isArray(data[key]); })) return {ok: false, err: "bad_data"};
  const files = Array.isArray(req.files) ? req.files : [];
  if (files.length > MAX_FILES) return {ok: false, err: "too_many_files"};
  for (let k = 0; k < files.length; k++) {
    const f = files[k];
    if (!f || !/^img\/u\/[a-z0-9]{6,40}\.jpg$/.test(f.path) || typeof f.b64 !== "string" || f.b64.length > MAX_B64 || !/^[A-Za-z0-9+\/=]+$/.test(f.b64)) return {ok: false, err: "bad_file"};
  }
  const lock = LockService.getScriptLock();
  lock.waitLock(60000);
  try {
    return commit(tok, data, files, req.base, req.force === true);
  } finally {
    lock.releaseLock();
  }
}

function gh(tok, method, path, body) {
  const o = {method: method, muteHttpExceptions: true, headers: {Authorization: "Bearer " + tok, Accept: "application/vnd.github+json", "X-GitHub-Api-Version": "2022-11-28"}};
  if (body) { o.contentType = "application/json"; o.payload = JSON.stringify(body); }
  const r = UrlFetchApp.fetch("https://api.github.com/repos/" + REPO + path, o), code = r.getResponseCode(), txt = r.getContentText();
  if (code >= 300) { const e = new Error("GitHub " + code + " " + method + " " + path + ": " + txt.slice(0, 200)); e.code = code; throw e; }
  return txt ? JSON.parse(txt) : null;
}
function ghText(tok, p, ref) {
  const r = UrlFetchApp.fetch("https://api.github.com/repos/" + REPO + "/contents/" + encodeURI(p) + "?ref=" + encodeURIComponent(ref || BRANCH), {muteHttpExceptions: true, headers: {Authorization: "Bearer " + tok, Accept: "application/vnd.github.raw+json", "X-GitHub-Api-Version": "2022-11-28"}});
  if (r.getResponseCode() === 404) return null;
  if (r.getResponseCode() >= 300) throw new Error("GitHub " + r.getResponseCode() + " read " + p);
  return r.getContentText("UTF-8");
}
function blobEntry(tok, path, content, encoding) {
  const blob = gh(tok, "post", "/git/blobs", {content: content, encoding: encoding});
  return {path: P(path), mode: "100644", type: "blob", sha: blob.sha};
}

// Сверяем ревизию в том же коммите, от которого строим дерево.
function commit(tok, data, files, expectedRev, force) {
  let photos;
  for (let attempt = 0; attempt < 4; attempt++) {
    const head = gh(tok, "get", "/git/ref/heads/" + BRANCH).object.sha;
    const cur = ghText(tok, P("data.js"), head) || "";
    const marker = cur.match(/(?:^|\n)\/\*rev:(\d+)\*\/\s*$/);
    const currentRev = marker ? +marker[1] : 0;
    if (!force && expectedRev !== currentRev) return {ok: false, err: "conflict", rev: currentRev};
    const rev = currentRev + 1;
    data.rev = rev;
    data.saved = new Date().toISOString();
    const js = "window.SITE_DATA = " + JSON.stringify(data).replace(/</g, "\\u003c") + ";\n/*rev:" + rev + "*/\n";
    if (js.length > MAX_JS) return {ok: false, err: "too_big"};

    if (!photos) photos = files.map(function (f) { return blobEntry(tok, f.path, f.b64, "base64"); });
    const tree = photos.concat(blobEntry(tok, "data.js", js, "utf-8"));
    const base = gh(tok, "get", "/git/commits/" + head).tree.sha;
    const nt = gh(tok, "post", "/git/trees", {base_tree: base, tree: tree});
    const nc = gh(tok, "post", "/git/commits", {
      message: "torty-shcherbinka: правки из режима мастера, версия " + rev,
      tree: nt.sha,
      parents: [head]
    });
    try {
      gh(tok, "patch", "/git/refs/heads/" + BRANCH, {sha: nc.sha, force: false});
      return {ok: true, rev: rev, commit: nc.sha};
    } catch (x) {
      if (attempt === 3 || (x.code !== 409 && x.code !== 422)) throw x;
      Utilities.sleep(1500);
    }
  }
}
